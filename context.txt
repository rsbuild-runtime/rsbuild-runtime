import React, { Suspense, lazy } from 'react';
import { Router, Switch, Route, Redirect } from 'react-router-dom';
import { createBrowserHistory } from 'history';
// @ts-nocheck
const history = createBrowserHistory();
export const rootContainer = (container: React.ReactElement | null) => {
  const routesTree = (
    <Router history={history}>
      <Switch>
        
        <Redirect key="/" exact={true} from="/" to="/home" />
        <Route
          key="route-1"
          
          exact={false}
          render={(props: unknown) => {
          const RawComponent = require('@/layouts').default;
          const wrappers: React.ComponentType<any>[] = [require('@/components/wrapper').default];
          const FinalComponent = wrappers.reduceRight((acc, Wrapper) => {
            return (p: Record<string, unknown>) => <Wrapper {...p}>{acc(p)}</Wrapper>;
          }, (p: Record<string, unknown>) => <RawComponent {...p} />);
          return (
            <Suspense fallback={null}>
              <FinalComponent {...(props as Record<string, unknown>)}>
            <Switch>
                <Route
                  key="/home"
                  path="/home"
                  exact={true}
                  render={(props: unknown) => {
                  const FinalComponent = require('@/pages/home').default;
                  return (
                    <Suspense fallback={null}>
                      <FinalComponent {...(props as Record<string, unknown>)}>
                      </FinalComponent>
                    </Suspense>
                  );
                  }}
                />
                <Route
                  key="/user"
                  path="/user"
                  exact={false}
                  render={(props: unknown) => {
                  const FinalComponent = (props: Record<string, unknown>) => <React.Fragment>{props.children}</React.Fragment>;
                  return (
                    <Suspense fallback={null}>
                      <FinalComponent {...(props as Record<string, unknown>)}>
                    <Switch>
                        <Route
                          key="/user/info"
                          path="/user/info"
                          exact={true}
                          render={(props: unknown) => {
                          const FinalComponent = lazy(() => import('@/pages/user/info'));
                          return (
                            <Suspense fallback={null}>
                              <FinalComponent {...(props as Record<string, unknown>)}>
                              </FinalComponent>
                            </Suspense>
                          );
                          }}
                        />
                        <Route
                          key="/user/dashboard"
                          path="/user/dashboard"
                          exact={true}
                          render={(props: unknown) => {
                          const FinalComponent = lazy(() => import('@/pages/user/dashboard'));
                          return (
                            <Suspense fallback={null}>
                              <FinalComponent {...(props as Record<string, unknown>)}>
                              </FinalComponent>
                            </Suspense>
                          );
                          }}
                        />
                    </Switch>
                      </FinalComponent>
                    </Suspense>
                  );
                  }}
                />
            </Switch>
              </FinalComponent>
            </Suspense>
          );
          }}
        />
      </Switch>
    </Router>
  );

  if (container) {
    return React.cloneElement(container, { children: routesTree });
  }

  return routesTree;
};
